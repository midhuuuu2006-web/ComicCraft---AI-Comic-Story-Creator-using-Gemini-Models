// ------------------------------------------------------------
// GEMINI
// ------------------------------------------------------------

const genAI = process.env.GEMINI_API_KEY
    ? new GoogleGenerativeAI(process.env.GEMINI_API_KEY)
    : null;

async function askGemini(prompt) {
    if (!genAI) {
        throw new Error("GEMINI_API_KEY is not configured.");
    }

    const model = genAI.getGenerativeModel({
        model: "gemini-2.0-flash"
    });

    const result = await model.generateContent(prompt);
    return result.response.text();
}

function extractJSON(text) {
    text = text.trim();

    if (text.startsWith("```")) {
        text = text.replace(/^```json/i, "");
        text = text.replace(/^```/i, "");
        text = text.replace(/```$/i, "");
    }

    const first = text.indexOf("{");
    const last = text.lastIndexOf("}");

    if (first !== -1 && last !== -1) {
        return JSON.parse(text.substring(first, last + 1));
    }

    const arrayFirst = text.indexOf("[");
    const arrayLast = text.lastIndexOf("]");

    if (arrayFirst !== -1 && arrayLast !== -1) {
        return JSON.parse(text.substring(arrayFirst, arrayLast + 1));
    }

    throw new Error("AI returned invalid JSON.");
}

app.get("/api/projects", (req, res) => {
    const projects = db.prepare(`
        SELECT id, title, genre, style, length, created_at, updated_at
        FROM projects
        ORDER BY updated_at DESC
    `).all();

    res.json(projects);
});

app.get("/api/projects/:id", (req, res) => {
    const project = db.prepare(`
        SELECT * FROM projects WHERE id = ?
    `).get(req.params.id);

    if (!project) {
        return res.status(404).json({
            error: "Project not found"
        });
    }

    project.characters = JSON.parse(project.characters || "[]");
    project.scenes = JSON.parse(project.scenes || "[]");
    project.panels = JSON.parse(project.panels || "[]");

    res.json(project);
});

app.post("/api/projects", (req, res) => {
    const {
        title,
        genre,
        idea,
        style,
        length
    } = req.body;

    if (!title) {
        return res.status(400).json({
            error: "Project title is required."
        });
    }

    const result = db.prepare(`
        INSERT INTO projects
        (title, genre, idea, style, length)
        VALUES (?, ?, ?, ?, ?)
    `).run(
        title,
        genre || "Adventure",
        idea || "",
        style || "Modern Comic",
        length || "5 pages"
    );

    res.json({
        id: result.lastInsertRowid
    });
});

app.put("/api/projects/:id", (req, res) => {
    const {
        title,
        genre,
        idea,
        style,
        length,
        story,
        characters,
        scenes,
        panels,
        dialogue
    } = req.body;

    db.prepare(`
        UPDATE projects
        SET
            title = ?,
            genre = ?,
            idea = ?,
            style = ?,
            length = ?,
            story = ?,
            characters = ?,
            scenes = ?,
            panels = ?,
            dialogue = ?,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
    `).run(
        title,
        genre,
        idea,
        style,
        length,
        story || "",
        JSON.stringify(characters || []),
        JSON.stringify(scenes || []),
        JSON.stringify(panels || []),
        dialogue || "",
        req.params.id
    );

    res.json({
        success: true
    });
});

app.delete("/api/projects/:id", (req, res) => {
    db.prepare(`
        DELETE FROM projects WHERE id = ?
    `).run(req.params.id);

    res.json({
        success: true
    });
});

app.post("/api/generate/story", async (req, res) => {
    try {
        const {
            idea,
            genre,
            style,
            length
        } = req.body;

        const prompt = `
You are a professional comic story writer.

Create a structured comic story.

Story idea:
${idea}

Genre:
${genre}

Comic style:
${style}

Length:
${length}

Return ONLY valid JSON in this format:

{
  "title": "",
  "premise": "",
  "setting": "",
  "mainConflict": "",
  "beginning": "",
  "middle": "",
  "ending": ""
}

Make the story creative, clear and suitable for a comic.
`;

        const answer = await askGemini(prompt);
        const story = extractJSON(answer);

        res.json(story);

    } catch (error) {
        console.error(error);

        res.status(500).json({
            error: error.message
        });
    }
});

async function api(url, options = {}) {

    const response = await fetch(
        url,
        {
            headers: {
                "Content-Type":
                    "application/json"
            },
            ...options
        }
    );

    const data = await response.json();

    if (!response.ok) {
        throw new Error(
            data.error || "Request failed"
        );
    }

    return data;
}

async function createProject() {

    const title =
        $("projectTitle").value.trim();

    const genre =
        $("projectGenre").value;

    const style =
        $("projectStyle").value;

    const length =
        $("projectLength").value;

    const idea =
        $("projectIdea").value.trim();

    if (!title || !idea) {

        toast(
            "Please enter title and story idea."
        );

        return;
    }

    try {

        const result =
            await api(
                "/api/projects",
                {
                    method: "POST",
                    body: JSON.stringify({
                        title,
                        genre,
                        idea,
                        style,
                        length
                    })
                }
            );

        currentProject = {
            id: result.id,
            title,
            genre,
            idea,
            style,
            length,
            story: "",
            characters: [],
            scenes: [],
            panels: [],
            dialogue: ""
        };

        toast("Project created!");

        openWorkspace();

    } catch (error) {

        toast(error.message);

    }
}

// SECURITY / HTML ESCAPE
// ------------------------------------------------------------

function escapeHtml(value) {

    if (
        value === null ||
        value === undefined
    ) {
        return "";
    }

    return String(value)
